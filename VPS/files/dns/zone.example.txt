; Example DNS zone for example.com (BIND format). Most DNS hosts let you import a file like this.
; Replace the IPs, provider hostnames and DKIM keys. TTL 300 while setting up; raise to 3600 later.
$ORIGIN example.com.
$TTL 3600

; ---- Web ------------------------------------------------------------------------------------
@                IN A       203.0.113.10          ; web-01 / load balancer IPv4
@                IN AAAA    2001:db8::10          ; IPv6
www              IN CNAME   example.com.
app              IN A       203.0.113.10
api              IN A       203.0.113.11          ; separate API server / LB
status           IN CNAME   statuspage.provider.example.   ; hosted status page

; ---- Certificate authority restriction -------------------------------------------------------
@                IN CAA     0 issue "letsencrypt.org"
@                IN CAA     0 issuewild "letsencrypt.org"
@                IN CAA     0 iodef "mailto:security@example.com"

; ---- Email: mailboxes at a hosted provider (replace with your provider's values) -------------
@                IN MX      10 mx1.mailprovider.com.
@                IN MX      20 mx2.mailprovider.com.
@                IN TXT     "v=spf1 include:_spf.mailprovider.com -all"
selector1._domainkey IN CNAME selector1.dkim.mailprovider.com.
_dmarc           IN TXT     "v=DMARC1; p=quarantine; rua=mailto:dmarc@example.com; adkim=s; aspf=s; pct=100"

; ---- Email: transactional sending from a subdomain (e.g. Amazon SES / Postmark) --------------
notify           IN MX      10 feedback-smtp.eu-west-1.amazonses.com.   ; SES custom MAIL FROM (bounces)
notify           IN TXT     "v=spf1 include:amazonses.com -all"
abc123._domainkey.notify IN CNAME abc123.dkim.amazonses.com.            ; SES gives 3 of these
_dmarc.notify    IN TXT     "v=DMARC1; p=reject; rua=mailto:dmarc@example.com"

; ---- Email: transport security + reporting --------------------------------------------------
_mta-sts         IN TXT     "v=STSv1; id=20261018"
mta-sts          IN A       203.0.113.10          ; serves /.well-known/mta-sts.txt over HTTPS
_smtp._tls       IN TXT     "v=TLSRPTv1; rua=mailto:tlsrpt@example.com"

; ---- Verification records (examples) --------------------------------------------------------
